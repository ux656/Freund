8. lib/mock-api.ts

ملاحظة: هذا هو نفس منطق المحاكاة من ملفك الأصلي، مع تحسينات بسيطة. لا يوجد تكامل واتساب حقيقي، وسيظل كذلك حتى تربط خادمًا حقيقيًا.

```ts
/**
 * ⚠️ MOCK API — للعرض فقط.
 * في الإنتاج: انقل هذا المنطق بالكامل إلى خادم (Express / Next.js Route Handler)
 * لأن أي مستخدم يمكنه التلاعب به عبر DevTools.
 */

export interface PairingResult {
  ok: boolean;
  sessionId?: string;
  code?: string;
  expiresAt?: number;
  serverTime?: number;
  nextAllowedAt?: number;
  error?: string;
  retryAfterMs?: number;
}

export interface SessionStatus {
  ok: boolean;
  status?: "awaiting" | "confirmed" | "expired";
  expiresAt?: number;
  serverTime?: number;
  error?: string;
}

const LIMITS = {
  cooldownMs: 60_000,
  perPhoneHourly: 3,
  perIPHourly: 4,
  codeTtlMs: 60_000,
  autoConfirmAfterMs: 8_000, // محاكاة webhook واتساب — للعرض فقط
  clockSkewWarnMs: 30_000,
};

const requestsByPhone = new Map<string, number[]>();
const requestsByIP = new Map<string, number[]>();
const sessions = new Map<
  string,
  { phone: string; code: string; expiresAt: number; createdAt: number; status: "awaiting" | "confirmed" | "expired" }
>();
const attemptsByPhone = new Map<string, number>();
const CLIENT_IP = "client-127.0.0.1";

function now() {
  return Date.now();
}
function delay(ms: number) {
  return new Promise((r) => setTimeout(r, ms));
}
function prune(map: Map<string, number[]>) {
  const cutoff = now() - 3_600_000;
  for (const [k, arr] of map) {
    const kept = arr.filter((t) => t > cutoff);
    if (kept.length) map.set(k, kept);
    else map.delete(k);
  }
}

export async function requestPairing({
  phone,
  captchaAnswer,
  captchaExpected,
}: {
  phone: string;
  captchaAnswer?: string;
  captchaExpected?: number;
}): Promise<PairingResult> {
  await delay(400);
  const t = now();
  const ip = CLIENT_IP;

  const attempts = attemptsByPhone.get(phone) || 0;
  if (attempts >= 1) {
    if (captchaAnswer === undefined || captchaAnswer === "") {
      return { ok: false, error: "captcha_required", serverTime: t };
    }
    if (Number(captchaAnswer) !== Number(captchaExpected)) {
      return { ok: false, error: "captcha_wrong", serverTime: t };
    }
  }

  prune(requestsByPhone);
  prune(requestsByIP);

  const phoneHistory = requestsByPhone.get(phone) || [];
  const lastReq = phoneHistory.length ? phoneHistory[phoneHistory.length - 1] : 0;
  if (t - lastReq < LIMITS.cooldownMs) {
    return {
      ok: false,
      error: "cooldown",
      retryAfterMs: LIMITS.cooldownMs - (t - lastReq),
      serverTime: t,
    };
  }

  if (phoneHistory.length >= LIMITS.perPhoneHourly) {
    return {
      ok: false,
      error: "phone_limit",
      retryAfterMs: 3_600_000 - (t - phoneHistory[0]),
      serverTime: t,
    };
  }

  const ipHistory = requestsByIP.get(ip) || [];
  if (ipHistory.length >= LIMITS.perIPHourly) {
    return {
      ok: false,
      error: "ip_limit",
      retryAfterMs: 3_600_000 - (t - ipHistory[0]),
      serverTime: t,
    };
  }

  const code = Array.from({ length: 8 }, () => Math.floor(Math.random() * 10)).join("");
  const sessionId = "sess_" + Math.random().toString(36).slice(2, 12);
  const expiresAt = t + LIMITS.codeTtlMs;

  phoneHistory.push(t);
  requestsByPhone.set(phone, phoneHistory);
  ipHistory.push(t);
  requestsByIP.set(ip, ipHistory);
  attemptsByPhone.set(phone, attempts + 1);
  sessions.set(sessionId, { phone, code, expiresAt, createdAt: t, status: "awaiting" });

  return {
    ok: true,
    sessionId,
    code,
    expiresAt,
    serverTime: t,
    nextAllowedAt: t + LIMITS.cooldownMs,
  };
}

export async function checkSession({ sessionId }: { sessionId: string }): Promise<SessionStatus> {
  await delay(300);
  const t = now();
  const s = sessions.get(sessionId);
  if (!s) return { ok: false, error: "session_not_found", serverTime: t };

  if (s.status === "awaiting" && t - s.createdAt >= LIMITS.autoConfirmAfterMs && t < s.expiresAt) {
    s.status = "confirmed";
  }
  if (s.status === "awaiting" && t >= s.expiresAt) {
    s.status = "expired";
  }
  return { ok: true, status: s.status, expiresAt: s.expiresAt, serverTime: t };
}

export const MOCK_LIMITS = LIMITS;
```

```tsx
"use client";

import { useEffect, useState } from "react";
import { Moon, Sun } from "lucide-react";
import { motion } from "framer-motion";

export default function ThemeToggle() {
  const [isDark, setIsDark] = useState(true);

  useEffect(() => {
    const stored = localStorage.getItem("theme");
    const prefersDark = !window.matchMedia("(prefers-color-scheme: light)").matches;
    const dark = stored === "dark" || (!stored && prefersDark);
    setIsDark(dark);
    document.documentElement.classList.toggle("dark", dark);
  }, []);

  const toggle = () => {
    const next = !isDark;
    setIsDark(next);
    document.documentElement.classList.toggle("dark", next);
    localStorage.setItem("theme", next ? "dark" : "light");
  };

  return (
    <button
      onClick={toggle}
      aria-label="تبديل الوضع"
```

14. components/PairingFlow.tsx

هذا هو المكوّن الأهم. يحافظ على نفس المنطق الأصلي (شاشات 1-2-3، CAPTCHA، cooldown، محاكاة التأكيد) لكن بواجهة زجاجية وتنقل ناعم.

```tsx
"use client";

import { useCallback, useEffect, useRef, useState } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { Copy, Check, Loader2, AlertCircle, Smartphone, KeyRound, ShieldCheck } from "lucide-react";
import { cn } from "@/lib/utils";
import { requestPairing, checkSession, MOCK_LIMITS } from "@/lib/mock-api";

type Step = 1 | 2 | 3;

export default function PairingFlow() {
  const [step, setStep] = useState<Step>(1);
  const [phone, setPhone] = useState("");
  const [sessionId, setSessionId] = useState<string | null>(null);
  const [code, setCode] = useState<string | null>(null);
  const [expiresAt, setExpiresAt] = useState(0);
  const [serverOffset, setServerOffset] = useState(0);
  const [cooldownUntil, setCooldownUntil] = useState(0);
  const [remainingSec, setRemainingSec] = useState(60);
  const [cooldownSec, setCooldownSec] = useState(0);
  const [verifyState, setVerifyState] = useState<"idle" | "awaiting" | "confirmed" | "expired">("idle");
  const [isVerifying, setIsVerifying] = useState(false);
  const [error, setError] = useState("");
  const [rateLimit, setRateLimit] = useState("");
  const [captchaExpected, setCaptchaExpected] = useState<number | null>(null);
  const [captchaA, setCaptchaA] = useState(0);
  const [captchaB, setCaptchaB] = useState(0);
  const [captchaInput, setCaptchaInput] = useState("");
  const [copied, setCopied] = useState(false);

  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const cooldownRef = useRef<ReturnType<typeof setInterval> | null>(null);

  const serverNow = () => Date.now() + serverOffset;
  const syncTime = (t?: number) => {
    if (typeof t === "number") setServerOffset(t - Date.now());
  };

  /* ===== Timer ===== */
  useEffect(() => {
    if (!sessionId) return;
    timerRef.current = setInterval(() => {
      const rem = Math.max(0, expiresAt - serverNow());
      setRemainingSec(Math.ceil(rem / 1000));
      if (rem <= 0) {
        clearInterval(timerRef.current!);
        if (verifyState === "awaiting") setVerifyState("expired");
      }
    }, 200);
    return () => clearInterval(timerRef.current!);
  }, [sessionId, expiresAt, verifyState]);

  /* ===== Cooldown ===== */
  useEffect(() => {
    if (!cooldownUntil) return;
    cooldownRef.current = setInterval(() => {
      const rem = Math.max(0, cooldownUntil - serverNow());
      setCooldownSec(Math.ceil(rem / 1000));
      if (rem <= 0) clearInterval(cooldownRef.current!);
    }, 500);
    return () => clearInterval(cooldownRef.current!);
  }, [cooldownUntil]);

  /* ===== Polling ===== */
  const stopPolling = useCallback(() => {
    if (pollRef.current) clearInterval(pollRef.current);
    pollRef.current = null;
  }, []);

  const verify = useCallback(
    async (auto = false) => {
      if (!sessionId || isVerifying) return;
      setIsVerifying(true);
      try {
        const res = await checkSession({ sessionId });
        syncTime(res.serverTime);
        if (!res.ok) {
          setVerifyState("expired");
          stopPolling();
          return;
        }
        if (res.status === "confirmed") {
          setVerifyState("confirmed");
          stopPolling();
          setTimeout(() => setStep(3), 800);
          return;
        }
        if (res.status === "expired") {
          setVerifyState("expired");
          stopPolling();
          return;
        }
        if (!auto) {
          setRateLimit("لم يتم التأكيد بعد. تأكد من إدخال الرمز في واتساب.");
        }
      } finally {
        setIsVerifying(false);
      }
    },
    [sessionId, isVerifying, stopPolling]
  );

  const startPolling = useCallback(() => {
    stopPolling();
    pollRef.current = setInterval(() => verify(true), 2000);
  }, [stopPolling, verify]);

  /* ===== Generate ===== */
  const handleGenerate = async () => {
    setError("");
    setRateLimit("");
    const res = await requestPairing({
      phone,
      captchaAnswer: captchaExpected !== null ? captchaInput : undefined,
      captchaExpected: captchaExpected ?? undefined,
    });
    syncTime(res.serverTime);

    if (res.ok) {
      setSessionId(res.sessionId!);
      setCode(res.code!);
      setExpiresAt(res.expiresAt!);
      setCooldownUntil(res.nextAllowedAt!);
      setVerifyState("awaiting");
      setCaptchaExpected(null);
      setCaptchaInput("");
      setStep(2);
      startPolling();
      return;
    }

    if (res.error === "captcha_required" || res.error === "captcha_wrong") {
      const a = Math.floor(Math.random() * 8) + 2;
      const b = Math.floor(Math.random() * 8) + 2;
      setCaptchaA(a);
      setCaptchaB(b);
      setCaptchaExpected(a + b);
      if (res.error === "captcha_wrong") setRateLimit("الإجابة غير صحيحة. حاول مرة أخرى.");
      return;
    }
    if (res.error === "cooldown") {
      setCooldownUntil(serverNow() + (res.retryAfterMs || 0));
      setRateLimit(`انتظر ${Math.ceil((res.retryAfterMs || 0) / 1000)} ثانية قبل طلب رمز جديد.`);
      return;
    }
    if (res.error === "phone_limit") {
      setRateLimit(`تجاوزت الحد المسموح (${MOCK_LIMITS.perPhoneHourly} رموز/ساعة لهذا الرقم).`);
      return;
    }
    if (res.error === "ip_limit") {
      setRateLimit("تم تقييد الشبكة مؤقتًا. حاول لاحقًا.");
      return;
    }
    setError("حدث خطأ غير متوقع. حاول لاحقًا.");
  };

  const copyCode = async () => {
    if (!code) return;
    try {
      await navigator.clipboard.writeText(code);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    } catch {}
  };

  const reset = () => {
    stopPolling();
    setStep(1);
    setPhone("");
    setSessionId(null);
    setCode(null);
    setExpiresAt(0);
    setCooldownUntil(0);
    setVerifyState("idle");
    setError("");
    setRateLimit("");
    setCaptchaExpected(null);
    setCaptchaInput("");
  };

  const canGenerate = phone.replace(/\D/g, "").length >= 10;

  return (
    <div className="space-y-5">
      {/* Steps */}
      <div className="flex items-center px-1">
        {[1, 2, 3].map((n, i) => (
          <div key={n} className="flex flex-1 items-center">
            <div className="flex shrink-0 items-center gap-2">
              <div
                className={cn(
                  "grid size-7 place-items-center rounded-full border text-[11px] font-semibold transition-all",
                  step === n && "border-primary bg-primary/10 text-primary",
                  step > n && "border-primary bg-primary text-bg",
                  step < n && "border-border text-text-3"
                )}
              >
                {step > n ? <Check size={13} strokeWidth={3} /> : n}
              </div>
              <span
                className={cn(
                  "text-[12px] font-medium whitespace-nowrap transition-colors",
                  step === n && "text-text",
                  step > n && "text-text-2",
                  step < n && "text-text-3"
                )}
              >
                {n === 1 ? "الرقم" : n === 2 ? "الاقتران" : "جاهز"}
              </span>
            </div>
            {i < 2 && (
              <div className="mx-2.5 h-0.5 flex-1 overflow-hidden rounded-full bg-border">
                <motion.div
                  className="h-full origin-right bg-primary"
                  initial={{ scaleX: 0 }}
                  animate={{ scaleX: step > n ? 1 : 0 }}
                  transition={{ duration: 0.5 }}
                />
              </div>
            )}
          </div>
        ))}
      </div>

      <AnimatePresence mode="wait">
        {/* ===== Screen 1 ===== */}
        {step === 1 && (
          <motion.div
            key="s1"
            initial={{ opacity: 0, y: 10 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -10 }}
            className="space-y-4"
          >
            <div className="lg lg-surface rounded-glass p-5">
              <label className="mb-2 block text-[11px] font-semibold uppercase tracking-wider text-text-2">
                رقم الهاتف مع كود الدولة
              </label>
              <div className="flex items-center overflow-hidden rounded-glass-sm border border-input bg-input" dir="ltr">
                <input
                  type="tel"
                  value={phone}
                  onChange={(e) => {
                    const digits = e.target.value.replace(/\D/g, "").slice(0, 15);
                    setPhone(digits);
                    setError("");
                  }}
                  placeholder="20xxxxxxxxx"
                  inputMode="numeric"
                  maxLength={16}
                  className="flex-1 bg-transparent px-4 py-3.5 font-body text-base font-medium tracking-wider text-text outline-none placeholder:text-text-3"
                />
                {phone && (
                  <button
                    onClick={() => setPhone("")}
                    className="mr-2 grid size-6 place-items-center rounded-full bg-surface-3 text-text-2 hover:bg-text-3 hover:text-text"
                    aria-label="مسح"
                  >
                    ✕
                  </button>
                )}
              </div>
              <p className="mt-2 flex items-center gap-1.5 text-[11.5px] text-text-3">
                <AlertCircle size={12} className="opacity-70" />
                اكتب كود الدولة ثم رقمك — بدون + أو مسافات
              </p>
              {error && (
                <p role="alert" className="mt-2 text-[11.5px] font-medium text-danger">
                  {error}
                </p>
              )}
            </div>

            {/* CAPTCHA */}
            {captchaExpected !== null && (
              <motion.div
                initial={{ opacity: 0, height: 0 }}
                animate={{ opacity: 1, height: "auto" }}
                className="lg lg-surface rounded-glass p-4"
              >
                <label className="mb-2 block text-[11px] font-semibold uppercase tracking-wider text-text-2">
                  تحقّق بشري مطلوب
                </label>
                <div className="mb-3 text-center font-display text-xl font-bold tracking-widest text-text" dir="ltr">
                  {captchaA} + {captchaB} = ?
                </div>
                <input
                  type="number"
                  value={captchaInput}
                  onChange={(e) => setCaptchaInput(e.target.value)}
                  onKeyDown={(e) => e.key === "Enter" && handleGenerate()}
                  placeholder="أدخل الإجابة"
                  className="w-full rounded-glass-sm border border-input bg-input px-4 py-3 text-center text-base font-semibold text-text outline-none focus:border-primary"
                />
              </motion.div>
            )}

            {rateLimit && (
              <p role="alert" className="rounded-glass-sm border border-danger/30 bg-danger/10 px-3.5 py-2.5 text-[12.5px] font-medium text-danger">
                {rateLimit}
              </p>
            )}

            <button
              onClick={handleGenerate}
              disabled={!canGenerate}
              className={cn(
                "relative w-full overflow-hidden rounded-glass-sm border border-[#B5543A] bg-gradient-to-b from-[#E5876B] via-primary to-primary-hover px-5 py-3.5 text-sm font-semibold text-white shadow-lg transition-all",
                "hover:-translate-y-px hover:shadow-xl active:translate-y-px disabled:cursor-not-allowed disabled:opacity-35 disabled:shadow-none"
              )}
            >
              توليد رمز الاقتران
            </button>
          </motion.div>
        )}

        {/* ===== Screen 2 ===== */}
        {step === 2 && (
          <motion.div
            key="s2"
            initial={{ opacity: 0, y: 10 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -10 }}
            className="space-y-4"
          >
            {/* Code panel */}
            <div
              onClick={copyCode}
              className={cn(
                "lg lg-surface cursor-pointer rounded-glass p-5 text-center transition-transform active:scale-[0.995]",
                !code && "cursor-default"
              )}
            >
              <div className="mb-3 flex items-center justify-center gap-1.5 text-[10px] font-semibold uppercase tracking-[0.14em] text-text-3">
                {copied ? <Check size={11} /> : <Copy size={11} />}
                {copied ? "تم النسخ!" : "رمز الاقتران المكوّن من 8 أرقام"}

              <div className="flex justify-center gap-1.5" dir="ltr">
                {code
                  ? code.split("").map((d, i) => (
                      <motion.span
                        key={i}
                        initial={{ opacity: 0, y: 8, scale: 0.85 }}
                        animate={{ opacity: 1, y: 0, scale: 1 }}
                        transition={{ delay: i * 0.04 }}

      </AnimatePresence>
```

```tsx
"use client";

import { useState } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { MessageSquare, X } from "lucide-react";

const responses: Record<string, string> = {
  link: "اذهب لصفحة الربط، أدخل رقمك، ثم أدخل رمز الاقتران في واتساب من: الإعدادات ← الأجهزة المرتبطة ← ربط جهاز.",

  commands: "يدعم البوت 5 أوامر أساسية حالياً. المزيد قادم قريباً.",
  human: "تواصل مع فريق التطوير من صفحة المطورين.",
};

export default function BotAssistant() {
  const [open, setOpen] = useState(false);
  const [messages, setMessages] = useState<{ role: "bot" | "user"; text: string }[]>([
    { role: "bot", text: "مرحباً! أنا مساعد ريبوت. كيف يمكنني مساعدتك؟" },
  ]);

  const ask = (action: string, label: string) => {
    setMessages((m) => [...m, { role: "user", text: label }]);
    setTimeout(() => {
      setMessages((m) => [...m, { role: "bot", text: responses[action] || "عذراً، لم أفهم." }]);
    }, 600);
  };

  return (
    <>
      <button
        onClick={() => setOpen(!open)}
        className="fixed bottom-6 left-6 z-50 grid size-14 place-items-center rounded-full bg-primary text-white shadow-xl transition-transform hover:scale-105"
        aria-label="مساعد ريبوت"
      >
        <MessageSquare size={22} />
      </button>

      <AnimatePresence>
        {open && (
          <motion.div
            initial={{ opacity: 0, y: 20, scale: 0.95 }}
            animate={{ opacity: 1, y: 0, scale: 1 }}
            exit={{ opacity: 0, y: 20, scale: 0.95 }}
            className="lg lg-surface fixed bottom-24 left-6 z-50 flex w-[360px] max-w-[calc(100vw-48px)] flex-col overflow-hidden rounded-glass"
          >
            <div className="flex items-center justify-between border-b border-border bg-surface-2 px-4 py-3">
              <div>
                <div className="text-sm font-semibold">Reboot</div>
                <div className="text-[10px] text-green">متصل الآن</div>
              </div>
              <button onClick={() => setOpen(false)} aria-label="إغلاق"><X size={18} /></button>
            </div>
            <div className="flex max-h-60 flex-col gap-2.5 overflow-y-auto bg-bg p-4">
              {messages.map((m, i) => (
                <div key={i} className={m.role === "bot" ? "max-w-[85%] self-start rounded-2xl rounded-br-sm border border-border bg-surface px-3.5 py-2.5 text-[13px] leading-relaxed" : "max-w-[85%] self-end rounded-2xl rounded-bl-sm bg-primary px-3.5 py-2.5 text-[13px] text-white"}>
                  {m.text}
                </div>
              ))}
            </div>
            <div className="flex flex-col gap-2 border-t border-border bg-surface-2 p-3">
              {[
                { action: "link", label: "كيف أربط الرقم؟" },
                { action: "security", label: "هل بياناتي آمنة؟" },
```

ملاحظات مهمة

· لا تغيير في الألوان: تم الحفاظ على نفس #141413، #D97757، #22c55e، إلخ.
· Liquid Glass حقيقي: استخدمت backdrop-filter: blur + saturate مع حدود متدرجة (rim gradient) وخلفية متحركة من blobs لظهور الانكسار، وليس مجرد bg-white/10.
· RTL/LTR: dir="rtl" افتراضيًا، مع استخدام خصائص Tailwind المنطقية (ms-, me-, start-, end-) حيث لزم.
· Accessibility: role="status" وaria-live على رسائل التحقق، role="alert" على Rate Limit، focus-visible واضح، ورفع تباين --danger إلى #FF8A80 في الوضع الداكن.
· المحاكاة محفوظة: lib/mock-api.ts هو نفس منطقك الأصلي، مع تعليق واضح بضرورة نقله لخادم حقيقي لاحقًا. لا يوجد تكامل واتساب حقيقي، ولم أحاول اختراعه.
· فصل المكونات: كل قسم في ملف مستقل، ويمكن استبدال mock-api بـ fetch حقيقي لاحقًا دون تغيير الواجهة.

هذا هو الحل الكامل. إذا واجهت أي خطأ في التثبيت أو التشغيل، أرسل لي نص الخطأ وسأساعدك في حله.
